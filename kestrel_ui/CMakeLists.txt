include(FetchContent)
set(KESTREL_LVGL_SOURCE_DIR "" CACHE PATH "Use an existing LVGL 9.2.2 checkout (offline builds)")
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
if(KESTREL_LVGL_SOURCE_DIR)
    add_subdirectory("${KESTREL_LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl")
else()
    # LVGL v9.2.2, pinned to its commit rather than a moving branch/tag.
    FetchContent_Declare(lvgl
        GIT_REPOSITORY https://github.com/lvgl/lvgl.git
        GIT_TAG 7f07a129e8d77f4984fff8e623fd5be18ff42e74)
    FetchContent_GetProperties(lvgl)
    if(NOT lvgl_POPULATED)
        FetchContent_Populate(lvgl)
        add_subdirectory("${lvgl_SOURCE_DIR}" "${lvgl_BINARY_DIR}")
    endif()
endif()

add_library(kestrel-ui-core STATIC stock_protocol.cpp stock_serial.cpp stock_backend.cpp model.cpp controller.cpp hud_data.cpp font_atlas.cpp input.cpp preferences.cpp ../hud_theme.cpp)
target_include_directories(kestrel-ui-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_options(kestrel-ui-core PRIVATE -Wall -Wextra -Wpedantic)

add_executable(kestrel-ui main.cpp display.cpp view.cpp perspective_text.cpp)
target_link_libraries(kestrel-ui PRIVATE kestrel-ui-core lvgl m)
target_compile_options(kestrel-ui PRIVATE -Wall -Wextra -Wpedantic)

add_executable(kestrel-ui-animation-bench animation_bench.cpp display.cpp view.cpp perspective_text.cpp)
target_link_libraries(kestrel-ui-animation-bench PRIVATE kestrel-ui-core lvgl m)
target_compile_options(kestrel-ui-animation-bench PRIVATE -Wall -Wextra -Wpedantic)

add_executable(kestrel-ui-assetgen assetgen.cpp)
target_include_directories(kestrel-ui-assetgen PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_options(kestrel-ui-assetgen PRIVATE -Wall -Wextra -Wpedantic)

option(KESTREL_UI_STATIC "Link the UI statically for the stock glibc runtime" OFF)
if(KESTREL_UI_STATIC)
    target_link_options(kestrel-ui PRIVATE -static)
    target_link_options(kestrel-ui-animation-bench PRIVATE -static)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(kestrel-ui-tests tests.cpp)
    target_link_libraries(kestrel-ui-tests PRIVATE kestrel-ui-core)
    target_compile_options(kestrel-ui-tests PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME kestrel-ui-core COMMAND kestrel-ui-tests)
    add_executable(kestrel-ui-serial-tests serial_tests.cpp)
    target_link_libraries(kestrel-ui-serial-tests PRIVATE kestrel-ui-core)
    target_compile_options(kestrel-ui-serial-tests PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME kestrel-ui-serial COMMAND kestrel-ui-serial-tests)
    add_executable(kestrel-ui-render-tests render_tests.cpp display.cpp view.cpp perspective_text.cpp)
    target_link_libraries(kestrel-ui-render-tests PRIVATE kestrel-ui-core lvgl m)
    target_compile_options(kestrel-ui-render-tests PRIVATE -Wall -Wextra -Wpedantic)
    set(KESTREL_UI_TEST_ASSETS "" CACHE PATH "Generated 1080p masks/font for rendering regression tests")
    if(KESTREL_UI_TEST_ASSETS)
        add_test(NAME kestrel-ui-render COMMAND kestrel-ui-render-tests "${KESTREL_UI_TEST_ASSETS}")
    endif()
endif()
install(TARGETS kestrel-ui RUNTIME DESTINATION bin)
